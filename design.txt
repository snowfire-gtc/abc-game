function genTerrain(){

  new THREE.Vector3(96,0,-6),new THREE.Vector3(90,0,20),new THREE.Vector3(94,0,44),
  new THREE.Vector3(102,0,66),new THREE.Vector3(96,0,88),new THREE.Vector3(78,0,104),

  let md=1e9;for(let i=0;i<pts.length-1;i++){
   const ax=pts[i].x,az=pts[i].z,bx=pts[i+1].x,bz=pts[i+1].z;
   const dx=bx-ax,dz=bz-az,L=dx*dx+dz*dz;
   let t=((x+.5-ax)*dx+(z+.5-az)*dz)/L;t=clamp(t,0,1);
   const px=ax+dx*t,pz=az+dz*t,d=dist2(x+.5,z+.5,px,pz);if(d<md)md=d;
  }

}

 if(t===T.WALK||t===T.DISC||t===T.PARK||t===T.PARKING||t===T.BPED)return 1;
 if(t===T.ROAD)return G.cross[i]?1:(mode==='jay'?4.5:-1);
 if(t===T.GRASS||t===T.SAND||t===T.SCORCH)return mode==='decent'?7:5;
 if(t===T.WATER)return mode==='swim'?10:-1;
 return -1;
}
function findPath(sx,sz,tx,tz,mode){
 if(!inMap(sx,sz)||!inMap(tx,tz))return null;
 const start=idx(sx,sz),goal=idx(tx,tz);
 if(start===goal)return[];
 const gc=cellCost(goal,mode);if(gc<0)return null;
 pathStamp++;heap.length=0;
 gScore[start]=0;gStamp[start]=pathStamp;came[start]=-1;hpush(0,start);
 let exp=0;
 while(heap.length){
  const[,cur]=hpop();
  if(closed[cur]===pathStamp)continue;closed[cur]=pathStamp;
  if(cur===goal){const path=[];let c=cur;while(c!==start){path.push(c);c=came[c]}path.reverse();return path}
  if(++exp>5200)break;
  const cx=cur%MAP,cz=(cur/MAP)|0;
  for(let d=0;d<4;d++){
   const nx=cx+DIR[d][0],nz=cz+DIR[d][1];if(!inMap(nx,nz))continue;
   const ni=nz*MAP+nx,cost=cellCost(ni,mode);if(cost<0)continue;
   if(closed[ni]===pathStamp)continue;
   const ng=(gStamp[cur]===pathStamp?gScore[cur]:1e9)+cost;
   if(gStamp[ni]!==pathStamp||ng<gScore[ni]){gScore[ni]=ng;gStamp[ni]=pathStamp;came[ni]=cur;
    hpush(ng+Math.abs(nx-tx)+Math.abs(nz-tz),ni)}
  }
 }
 return null;
}
function walkMode(l){return l.traits.criminal?'swim':l.traits.decent?'decent':'jay'}
function navTo(l,tx,tz,then){
 const sx=Math.floor(l.px),sz=Math.floor(l.pz);
 const p=findPath(sx,sz,tx,tz,walkMode(l));
 if(p===null)return false;
 l.path=p;l.pi=0;l.target=then;return true;
}

  return null;

 }
 if(id==='bped'||id==='broad'){

 }

function tryPlace(x,z){
 const item=BUILD_ITEMS.find(i=>i.id===mode.build);if(!item)return;
 const err=placementError(item,x,z);
 if(err){if(!mouse.paint){toast(err,'err');SFX.err()}return}
 const id=item.id;
 const spend=(c)=>{if(G.tokens<c){if(!mouse.paint){toast('Не хватает токенов: '+fmt(c)+'₮','err');SFX.err()}return false}G.tokens-=c;flashChip('chipTokens');return true};
 if(id==='demo'){
  const i=idx(x,z);
  const obj=G.objects.find(o=>o.x===x&&o.z===z);
  const bi=G.occ[i];
  if(bi>=0){if(!spend(1))return;const b=getB(bi);removeBuilding(b);SFX.place();news('🧨','Снесено: '+bName(b.type)+'.',0);return}
  if(obj){if(!spend(1))return;removeObject(obj);SFX.place();return}
  const t=G.tiles[i];
  if(t===T.BPED||t===T.BROAD){if(!spend(1))return;G.tiles[i]=T.WATER;removeBridgeMesh(x,z);G.navVer++;refreshTiles();SFX.place();return}
  if(G.cross[i]){if(!spend(1))return;G.cross[i]=0;refreshTiles();SFX.place();return}
  const tr=G.trees.findIndex(t=>t.x===x&&t.z===z);
  if(tr>=0){if(!spend(1))return;G.trees.splice(tr,1);refreshTrees();SFX.place();return}
  if(t!==T.GRASS){if(!spend(1))return;G.tiles[i]=T.GRASS;G.navVer++;refreshTiles();SFX.place();return}
  return;
 }
 if(id==='grass'||id==='water'||id==='spruce'){
  const i=idx(x,z);
  if(id==='spruce'){if(G.trees.some(t=>t.x===x&&t.z===z))return;if(!spend(5))return;addTree(x,z);refreshTrees();SFX.place();return}
  const target=id==='grass'?T.GRASS:T.WATER;
  if(G.tiles[i]===target)return;

  cells.forEach(([cx,cz])=>{G.tiles[idx(cx,cz)]=T.ROAD;G.trees=G.trees.filter(t=>t.x!==cx||t.z!==cz)});
  G.navVer++;refreshTiles();refreshTrees();SFX.place();return}
 if(id==='bped'||id==='broad'){
  const cells=mode.orient%2?[[x,z],[x,z+1]]:[[x,z],[x+1,z]];
  if(!spend(50))return;
  cells.forEach(([cx,cz])=>{G.tiles[idx(cx,cz)]=id==='bped'?T.BPED:T.BROAD;addBridgeMesh(cx,cz,id==='bped'?'ped':'road',mode.orient)});

  news('🌉','Новый мост через реку! Речные трамваи проплывают под ним.',0);return}
 if(id==='cross'){
  const cells=mode.orient%2?[[x,z],[x,z+1]]:[[x,z],[x+1,z]];
  if(!spend(40))return;

 if(id==='parking'){

 const b=placeBuilding(id,x,z,mode.orient,id==='print'?mode.level:1);

  const cells=[];
  if(b.orient===0){for(let cz=b.z;cz<b.z+3;cz++)cells.push([b.x+b.w,cz]);for(let i=0;i<nw;i++)cells.push([b.x+i,b.z+3])}
  else if(b.orient===1){for(let cx=b.x;cx<b.x+3;cx++)cells.push([cx,b.z+b.d]);for(let i=0;i<nw;i++)cells.push([b.x-1,b.z+i])}
  else if(b.orient===2){for(let cz=b.z;cz<b.z+3;cz++)cells.push([b.x-1,cz]);for(let i=0;i<nw;i++)cells.push([b.x+b.w-1-i,b.z-1])}
  else{for(let cx=b.x;cx<b.x+3;cx++)cells.push([cx,b.z-1]);for(let i=0;i<nw;i++)cells.push([b.x+3,b.z+b.d-1-i])}
  for(const[cx,cz]of cells){
   if(!inMap(cx,cz))return'край';
   const i=idx(cx,cz),t=G.tiles[i];
   if(G.occ[i]>=0&&G.occ[i]!==b.id)return'занято';
   if(t===T.WATER||t===T.ROAD)return'вода/дорога';
  }
 }
 return null;
}

function frame(t){
 raf=requestAnimationFrame(frame);
 let dt=Math.min(.1,(t-lastT)/1000||.016);lastT=t;
 if(!G)return;
 const spd=G.speed;
 const gdt=dt*spd;
 if(spd>0&&!G.over){
  timeTick(gdt*15); // 15 game min per real sec
  const dtH=gdt*15/60;
  // letters
  for(const l of G.letters){
   if(l.state==='patient'||l.state==='ride'||l.state==='carRide'||l.state==='transport')continue;
   if(l.path&&l.pi<l.path.length){
    const arrived=moveEntity(l,gdt,l.traits.stupid&&Math.random()<.02?.4:1.25*(stage(l)==='baby'?.6:stage(l)!=='adult'?.85:1),false);
    if(l.mesh){
     l.walkPhase+=gdt*9;
     const bob=Math.abs(Math.sin(l.walkPhase))*.09;
     const swim=G.tiles[idx(clamp(Math.floor(l.px),0,127),clamp(Math.floor(l.pz),0,127))]===T.WATER;
     l.mesh.position.set(l.px,swim?-.22:bob,l.pz);
     if(l.angle!=null)l.mesh.rotation.y=lerpAngle(l.mesh.rotation.y,l.angle,gdt*8);
     l.mesh.rotation.z=Math.sin(l.walkPhase)*.06;
     if(l.legs)l.legs.rotation.x=Math.sin(l.walkPhase)*.4;
     if(swim&&Math.random()<.1)splash(l.px,l.pz);
    }
    if(arrived)letterArrive(l);
   }else if(l.mesh){
    if(l.hopT>0){l.hopT-=gdt;l.mesh.position.y=Math.sin((0.4-l.hopT)/.4*Math.PI)*.35}
    else if(l.state==='working'){l.mesh.position.y=Math.abs(Math.sin(t/300+l.id))*.04}
    else l.mesh.position.y=lerp(l.mesh.position.y,0,gdt*6);
    l.mesh.rotation.z=lerp(l.mesh.rotation.z,0,gdt*5);
   }
   if(l.state==='sleep'&&l.mesh){l.mesh.visible=G.hour<6.5||G.hour>=21.5?true:true}
  }
  simAcc+=gdt;
  if(simAcc>=.5){
   const step=simAcc;simAcc=0;
   for(const l of G.letters)letterThink(l,step*15/60);
   jobTick();
  }
  printTick(dtH);
  fireTick(dtH);
  meteorTick(gdt);
  for(const v of G.vehicles)vehicleThink(v,gdt);
  for(const r of G.rabbits)rabbitTick(r,gdt);
  // fountain jets
  for(const b of G.buildings){
   if(b.type==='fountain'&&Math.random()<.35)spawnS(b.x+1+rnd(-.1,.1),1.1,b.z+1+rnd(-.1,.1),rnd(-.5,.5),rnd(1.5,2.5),rnd(-.5,.5),'#7fd0ff',rnd(.4,.8));
   if((b.type==='house'||b.type==='tower')&&(G.hour<8||G.hour>19)&&Math.random()<.02){
    const m=buildModel(b.type,0,b.variant).meta;
    if(m.chimney){let[cx,cy,cz]=m.chimney;
     for(let r=0;r<b.orient;r++){const t2=cx;cx=-cz;cz=t2}
     spawnP(b.x+b.w/2+cx,cy,b.z+b.d/2+cz,rnd(-.1,.2),rnd(.4,.8),rnd(-.1,.1),'#9aa4b8',rnd(1.2,2),.4)}
   }
  }
  particlesTick(gdt);
 }else{particlesTick(dt*.2)}
 updateSky(dt);
 applyTiles();applyTrees();
 updateCamera(dt);
 // selection refresh
 uiAcc+=dt;
 if(uiAcc>.45){uiAcc=0;
  $('resTokens').textContent=fmt(G.tokens);
  $('resPop').textContent=G.letters.length+'/'+housingCap();
  $('resWordsDay').textContent=G.wordsToday;
  $('resWordsTotal').textContent=fmt(G.wordsTotal);
  $('timeText').textContent=fmtTime();
  $('dayText').textContent='День '+G.day;
  $('timeIcon').textContent=G.weather.rain?'☔':G.hour<6||G.hour>=22?'🌙':G.hour<11?'🌅':G.hour<17?'☀':'🌇';
  $('todayStats').innerHTML='сегодня: <b>'+G.wordsToday+'</b> слов · <b>+'+fmt(G.tokensToday)+'₮</b>';
  if(selected)renderSel();
  document.querySelectorAll('.bcard').forEach(c=>{
   const item=BUILD_ITEMS.find(i=>i.id===c.dataset.item);
   const price=item.id==='print'?256*Math.pow(2,mode.level-1):item.price;
   c.classList.toggle('cant',G.tokens<price)});
 }
 saveAcc+=dt;
 if(saveAcc>30){saveAcc=0;if(G.settings.autosave&&!G.over)saveGame()}
 renderer.render(scene,camera);
}
